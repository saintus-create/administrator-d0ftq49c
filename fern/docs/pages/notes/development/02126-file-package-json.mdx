// FILE: src/app/api/generate-app/route.ts
// This backend API route is the powerful engine. Its logic is sound and remains unchanged.
import \{ NextResponse \} from 'next/server';
import \{ auth \} from '@/lib/auth';
import \{ db \} from '@/lib/db';
import OpenAI from 'openai';

const openai = new OpenAI(\{
  apiKey: process.env.OPENAI_API_KEY,
\});

const PLANNER_PROMPT = `
You are an expert full-stack software architect. A user wants to build an application.
Your task is to deconstruct their request and create a detailed, actionable plan for an AI code generator.
The user's prompt is: "\{USER_PROMPT\}"

Based on this prompt, you must:
1.  Invent a short, descriptive, kebab-case name for the application (e.g., "knowledge-base", "todo-app").
2.  Choose the most appropriate tech stack. Always use: "Next.js App Router", "TypeScript", "Tailwind CSS".
3.  Define the complete file structure needed for this application. Be specific. Include everything from package.json to component files, content files, and CSS.
4.  For each file, write a "generation_prompt" that an AI code generator (like GPT-4) can use to create the file's content. This prompt should be extremely clear, specifying the exact libraries, functions, and styling to use. For code files, demand only raw code output. For content files (like .mdx), demand realistic content based on the user's prompt.

Output your response as a single, valid JSON object. Do not include any text or markdown formatting before or after the JSON.
`;

export async function POST(request: Request) \{
  const session = await auth();
  if (!session?.user?.id) \{
    return NextResponse.json(\{ error: 'Unauthorized' \}, \{ status: 401 \});
  \}

  const \{ prompt \} = await request.json();
  if (!prompt) \{
    return NextResponse.json(\{ error: 'Prompt is required' \}, \{ status: 400 \});
  \}

  try \{
    const plannerResponse = await openai.chat.completions.create(\{
      model: 'gpt-4-turbo-preview',
      messages: [\{ role: 'user', content: PLANNER_PROMPT.replace('\{USER_PROMPT\}', prompt) \}],
      response_format: \{ type: 'json_object' \},
    \});

    const plan = JSON.parse(plannerResponse.choices[0].message.content || '\{\}');
    if (!plan.appName || !plan.filePlan) \{
      throw new Error('AI failed to generate a valid plan.');
    \}

    const project = await db.project.create(\{
      data: \{
        name: plan.appName,
        prompt: prompt,
        userId: session.user.id,
      \},
    \});

    const fileGenerationPromises = plan.filePlan.map(async (fileToGen: \{ path: string; generation_prompt: string \}) =\> \{
      const fileContentResponse = await openai.chat.completions.create(\{
        model: 'gpt-4-turbo-preview',
        messages: [\{ role: 'user', content: `Generate the content for the file path "$\{fileToGen.path\}". Strictly output only the raw content for the file, with no explanation, introduction, or markdown formatting. The requirements are: $\{fileToGen.generation_prompt\}` \}],
      \});
      const content = fileContentResponse.choices[0].message.content || '';
      return \{
        path: fileToGen.path,
        content: content,
        projectId: project.id,
      \};
    \});

    const generatedFiles = await Promise.all(fileGenerationPromises);

    await db.file.createMany(\{
      data: generatedFiles,
    \});

    return NextResponse.json(\{ projectId: project.id \});

  \} catch (error) \{
    console.error('Error during app generation:', error);
    return NextResponse.json(\{ error: 'Failed to generate application.' \}, \{ status: 500 \});
  \}
\}

interface Message \{
  role: 'user' | 'ai' | 'status';
  content: string;

export function ChatInterface() \{
  const [messages, setMessages] = useState\<Message[]\>([
    \{ role: 'ai', content: "What application would you like to build today? Be as descriptive as you'd like." \}
  ]);
  const [input, setInput] = useState("");
  const [isLoading, setIsLoading] = useState(false);
  const router = useRouter();
  const messagesEndRef = useRef\<HTMLDivElement\>(null);

  useEffect(() =\> \{
    messagesEndRef.current?.scrollIntoView(\{ behavior: "smooth" \});
  \}, [messages]);

  const handleSubmit = async (e: React.FormEvent) =\> \{
    e.preventDefault();
    if (!input || isLoading) return;

    const userMessage: Message = \{ role: 'user', content: input \};
    const statusMessage: Message = \{ role: 'status', content: 'Thinking...' \};
    setMessages(prev =\> [...prev, userMessage, statusMessage]);
    setInput("");
    setIsLoading(true);

    try \{
      const response = await fetch('/api/generate-app', \{
        method: 'POST',
        headers: \{ 'Content-Type': 'application/json' \},
        body: JSON.stringify(\{ prompt: input \}),
      \});

      if (!response.ok) \{
        const errorData = await response.json();
        throw new Error(errorData.error || 'An unknown error occurred.');
      \}

      const \{ projectId \} = await response.json();
      const successMessage: Message = \{ role: 'ai', content: `I've finished generating your application. You can view the files here: [View Project](/p/$\{projectId\})` \};
      setMessages(prev =\> [...prev.slice(0, -1), successMessage]);
      router.refresh(); // Refresh sidebar with new project

    \} catch (err: any) \{
      const errorMessage: Message = \{ role: 'ai', content: `I'm sorry, I ran into an error: $\{err.message\}` \};

      setIsLoading(false);
    \}
  \};
