# AI Frontier: Autonomous AI-Driven Website with CMS

An autonomous web application powered by a Node.js CLI and Next.js UI, with a CMS for content management, leveraging multi-model AI orchestration, context management, checkpointing, runtime debugging, MCP extensibility, and self-auditing.

## Setup
1. Clone: `git clone `
2. Copy `.env.example` to `.env` and add API keys (GROK_API_KEY from https://x.ai/api, DEEPSEEK_API_KEY from https://platform.deepseek.com/).
3. Install: `npm install`
4. Build: `npm run build`
5. Run CLI: `npm start -- --prompt "Generate a blog post"` (headless) or `npm start` (interactive, opens UI at http://localhost:3000).
6. Run web only: `npm run web`

## Features
- **Autonomous Content Generation**: Uses DeepSeek/Grok to generate blog posts/pages, triggered hourly via scheduler.
- **CMS Integration**: Create/edit/publish content via CLI (`cms:init`, `cms:create`) or admin UI (`/cms`) with AI-enhanced SEO and suggestions.
- **Self-Auditing**: 8-step AI code auditing pipeline (tokenization, AST, static analysis, embeddings).
- **Context Management**: Loads files/folders/URLs with file watching.
- **Checkpointing**: Git-integrated state preservation.
- **Runtime Debugging**: Puppeteer and terminal command execution.
- **MCP Extensibility**: Modular tool integration.
- **Next.js UI**: Responsive, dark/light mode interface with content display and CMS admin.

## Usage
- **Interactive**: Open http://localhost:3000 for main UI, http://localhost:3000/cms for CMS admin.
- **Headless**: `--prompt "task"` for generation, `--audit` for code auditing, `cms:init` or `cms:create ` for CMS.
- **Options**: `--model grok` (default), `--debug` for runtime checks.
- **Tools**: Via MCP, e.g., `bashExec`, `auditCode`, `cmsDeploy`.

## Notes
- Requires API keys in .env.
- CMS admin requires authentication (default: admin/secret; update in production).
- Autonomous scheduler runs content generation every hour.
- Security: Input sanitization, path traversal checks, JWT auth for CMS.
- If API calls fail, check keys and network.

import * as fs from "fs-extra";
import * as path from "path";

export class ContextManager \{
  private context: string[] = [];

  constructor(private directory: string) \{
    if (!fs.existsSync(directory)) throw new Error(`Directory $\{directory\} does not exist`);
    this.loadContext();
    this.watchDirectory();
  \}

  private loadContext() \{
    try \{
      const files = fs.readdirSync(this.directory);
      this.context = files
        .filter((file) =\> file.endsWith(".ts") || file.endsWith(".tsx"))
        .map((file) =\> fs.readFileSync(path.join(this.directory, file), "utf-8"));
    \} catch (error) \{
      console.warn("Error loading context:", error);
      this.context = [];
    \}
  \}

  private watchDirectory() \{
    try \{
      fs.watch(this.directory, (event, filename) =\> \{
        if (filename && (filename.endsWith(".ts") || filename.endsWith(".tsx"))) \{
          this.loadContext();
        \}
      \});
    \} catch (error) \{
      console.warn("Error setting up file watcher:", error);
    \}
  \}

  getContext(): string \{
    return this.context.join("\n");
  \}
\}

import * as parser from "@babel/parser";
import traverse from "@babel/traverse";
import LRU from "lru-cache";

export class CodeAuditor \{
  private cache = new LRU(\{ max: 100 \});

  async audit(context: string, code: string): Promise\<\{ code: string; issues: \{ type: string; message: string; loc?: any \}[] \}\> \{
    if (!code) throw new Error("Code is required");
    const cached = this.cache.get(code);
    if (cached) return cached;

    const issues: \{ type: string; message: string; loc?: any \}[] = [];
    try \{
      const ast = parser.parse(code, \{ sourceType: "module", plugins: ["typescript", "jsx"] \});
      traverse(ast, \{
        CallExpression(path) \{
          if (path.node.callee.type === "Identifier" && path.node.callee.name === "eval") \{
            issues.push(\{ type: "security", message: "Use of eval detected", loc: path.node.loc \});
          \}
        \}
      \});
    \} catch (error) \{
      issues.push(\{ type: "syntax", message: `Syntax error: $\{String(error)\}` \});
    \}

    const result = \{ code, issues \};
    this.cache.set(code, result);
    return result;
  \}
\}

export class WebsiteCMS \{
  private templates: Record = \{\};
  private contentStore: Record = \{\};

  constructor(private projectRoot: string) \{\}

  async init() \{
    try \{
      await this.loadTemplates();
      await this.loadContent();
    \} catch (error) \{

  async loadTemplates() \{
    const templatePath = path.join(this.projectRoot, "web/cms/templates");
    try \{
      await fs.ensureDir(templatePath);
      const files = await fs.readdir(templatePath);
      for (const file of files) \{
        const name = path.basename(file, ".html");
        this.templates[name] = await fs.readFile(path.join(templatePath, file), "utf-8");
      \}
      if (Object.keys(this.templates).length === 0) \{
        this.templates["default"] = `\{\{content\}\}`;
      \}
    \} catch (error) \{
      console.warn("No templates found, using default:", error);
      this.templates["default"] = `\{\{content\}\}`;
    \}
  \}

  async loadContent() \{
    const contentPath = path.join(this.projectRoot, "web/content");
    try \{
      await fs.ensureDir(contentPath);
      const files = await fs.readdir(contentPath);
      for (const file of files) \{
        if (file.endsWith(".json")) \{
          const data = await fs.readJson(path.join(contentPath, file));
          this.contentStore[file] = data;
        \}
      \}
    \} catch (error) \{
      console.warn("No content found:", error);
    \}
  \}

  async createPage(pageType: string, content: any): Promise \{
    if (!pageType || !content) throw new Error("Page type and content are required");
    const template = this.templates[pageType] || this.templates["default"];
    if (!template) throw new Error(`No template found for $\{pageType\}`);
    const sanitizedContent = this.sanitizeContent(content);
    const rendered = this.renderTemplate(template, sanitizedContent);
    const slug = sanitizedContent.slug || `page-$\{Date.now()\}`;
    const pagePath = `web/content/$\{slug\}.html`;
    const jsonPath = `web/content/$\{slug\}.json`;
    try \{
      await fs.ensureDir(path.join(this.projectRoot, "web/content"));
      await fs.writeFile(path.join(this.projectRoot, pagePath), rendered);
      await fs.writeJson(path.join(this.projectRoot, jsonPath), sanitizedContent);
      this.contentStore[`$\{slug\}.json`] = sanitizedContent;
      return pagePath;
    \} catch (error) \{
      console.error("Page creation failed:", error);
      throw error;
    \}
  \}

  private sanitizeContent(content: any): any \{
    const sanitized: any = \{\};
    for (const key in content) \{
      if (typeof content[key] === "string") \{
        sanitized[key] = sanitizeHtml(content[key], \{
          allowedTags: sanitizeHtml.defaults.allowedTags.concat(["img"]),
          allowedAttributes: \{ img: ["src", "alt"], a: ["href"] \},
        \});
      \} else \{
        sanitized[key] = content[key];
      \}
    \}
    return sanitized;
  \}

  private renderTemplate(template: string, data: any): string \{
    return template.replace(/\\{\\{(\w+)\\}\\}/g, (_, key) =\> data[key] || "");
  \}

  getContent(slug: string): any \{
    return this.contentStore[`$\{slug\}.json`] || null;
  \}
