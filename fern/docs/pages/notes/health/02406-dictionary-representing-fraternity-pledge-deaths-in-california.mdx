# Function to display the data
def display_fraternity_data(data):
    print("Fraternity Pledge Deaths in California (1957�2019):")
    print("-" * 50)
    total_deaths = 0
    oversight_counts = \{"IFC": 0, "NPHC": 0, "Local": 0, "Local/Uncertain": 0\}

    for chapter, info in data.items():
        tally = info["tally"]
        oversight = info["oversight"]
        details = info["details"]
        
        print(f"Chapter: \{chapter\}")
        print(f"  Tally: \{tally\}")
        print(f"  Oversight: \{oversight\}")
        if tally == 2:  # Special case for Alpha Kappa Alpha
            print(f"  Date: \{details['date']\}")
            print(f"  Names: \{', '.join(details['names'])\}")
            print(f"  Cause: \{details['cause']\}")
        else:
            print(f"  Date: \{details['date']\}")
            print(f"  Name: \{details['name']\}")
            print(f"  Cause: \{details['cause']\}")
        print()

        total_deaths += tally
        oversight_counts[oversight] = oversight_counts.get(oversight, 0) + tally

    # Summary
    print("-" * 50)
    print(f"Total Deaths: \{total_deaths\}")
    print(f"Total Chapters: \{len(data)\}")
    print("\nBreakdown by Oversight:")
    for oversight, count in oversight_counts.items():
        if count \> 0:
            print(f"  \{oversight\}: \{count\} deaths")

# Run the display function
if __name__ == "__main__":
    display_fraternity_data(fraternity_deaths)
