import collections

class BalancedNarrativeAnalyzer:
    """
    A class designed to analyze narratives while minimizing bias through balanced evaluation.
    Uses systematic feature extraction and scoring mechanisms.
    """
    
    def __init__(self):
        # Define neutral patterns for analysis
        self.patterns = \{
            'temporal': r'\b(?:at|on|during|from)\s+\d\{1,2\}-\d\{1,2\}\b',
            'frequency': r'(?:daily|weekly|monthly|regularly)',
            'contact': r'(?:email|call|message|contact)',
            'threat': r'(?:threaten|warn|state|claim)',
            'action': r'(?:take|make|do|perform)'
        \}
        
        # Initialize counters for balanced tracking
        self.feature_counts = collections.defaultdict(int)
        self.context_counts = collections.defaultdict(lambda: collections.Counter())
        
    def preprocess_text(self, text: str) -\> str:
        """
        Clean and normalize text for analysis.
        Removes extra whitespace and converts to lowercase.
        """
        return ' '.join(text.lower().split())

    def extract_features(self, text: str) -\> Dict[str, List[Tuple[int, str]]]:
        """
        Extract features while tracking context and frequency.
        Returns dictionary mapping pattern types to their occurrences.
        """
        cleaned_text = self.preprocess_text(text)
        features = \{\}
        
        for pattern_type, pattern in self.patterns.items():
            matches = [(m.start(), m.group()) 
                      for m in re.finditer(pattern, cleaned_text)]
            
            if matches:
                features[pattern_type] = matches
                self.feature_counts[pattern_type] += len(matches)
                
                # Track contextual information
                for pos, match in matches:
                    context = cleaned_text[max(0, pos-20):min(len(cleaned_text), pos+20)]
                    self.context_counts[pattern_type][context] += 1
                    
        return features

    def analyze_narrative(self, text: str) -\> Dict:
        """
        Perform comprehensive narrative analysis with balanced scoring.
        Returns detailed analysis including feature frequencies and contexts.
        """
        features = self.extract_features(text)
        
        # Calculate balanced scores based on pattern diversity
        total_patterns = sum(self.patterns.values())
        diversity_score = len(features) / len(self.patterns)
        
        # Calculate context variety score
        context_scores = \{\}
        for pattern_type, contexts in self.context_counts.items():
            total_occurrences = sum(contexts.values())
            unique_contexts = len(contexts)
            context_scores[pattern_type] = unique_contexts / total_occurrences if total_occurrences \> 0 else 0
        
        return \{
            'features': features,
            'diversity_score': diversity_score,
            'context_variety': context_scores,
            'pattern_frequencies': dict(self.feature_counts)
        \}

        \}

        for pattern_type, pattern in self.meta_patterns.items():

            if matches:
                meta_features[pattern_type] = matches
                self.pattern_usage_counts[pattern_type] += len(matches)
                
        return meta_features

    def evaluate_analysis_quality(self, features: Dict, meta_features: Dict) -\> Dict:
        """
        Evaluate the quality of the current analysis based on meta-criteria.
        Returns metrics about analysis balance and potential bias.
        """
        # Calculate meta-metrics
        total_patterns = sum(len(v) for v in features.values())
        meta_pattern_count = sum(len(v) for v in meta_features.values())
        
        # Coverage metric (how well do we cover different pattern types?)
        coverage_score = len(features) / len(self.patterns)
        
        # Balance metric (distribution across pattern categories)
        pattern_distribution = \{k: len(v)/total_patterns 
                              for k, v in features.items()\}
        balance_score = 1 - max(pattern_distribution.values()) if total_patterns \> 0 else 0
        
        # Consistency metric (stability of pattern usage over time)
        if self.analysis_history:
            historical_avg = \{
                k: sum(d[k] for d in self.analysis_history) / len(self.analysis_history)
                for k in self.patterns.keys()
            \}
            consistency_score = sum(abs(historical_avg[k] - pattern_distribution.get(k, 0))
                                  for k in self.patterns.keys()) / len(self.patterns)
        else:
            consistency_score = 1.0
            
        # Store metrics for trend analysis
        self.evaluation_metrics['meta_coverage'].append(coverage_score)
        self.evaluation_metrics['pattern_balance'].append(balance_score)
        self.evaluation_metrics['self_consistency'].append(consistency_score)
        
        return \{
            'coverage_score': coverage_score,
            'balance_score': balance_score,
            'consistency_score': consistency_score,
            'meta_pattern_ratio': meta_pattern_count / total_patterns if total_patterns \> 0 else 0\}
